%% ============================================================
%%  MANUAL PIPELINE (full control, no scripts)
%% ============================================================

addpath(genpath(fullfile(pwd, 'src')));

% 1. Load from run file
[scenario, config, sensors, metas] = trackbench.config.loadRunFile("my_run");

% 2. Generate detections
dataLog = trackbench.detections.runDetections(scenario, config.degradation.enabled, metas, config.environment);

% 3. Plot the scene
trackbench.reporting.plotInitialScenario(dataLog, false);   % false = static

% 4. Build a tracker
params = config.active_params;
numSensors = numel(sensors.tower);
tracker = trackbench.tracking.buildTracker('GNN', 'IMM', params, ...
    config.tracker_global, config.filter_params, params.pd, numSensors);

% 5. Run tracker + get metrics
[trackSummary, truthSummary, trackMetrics, truthMetrics, ...
    tTime, assignLog, swapReport] = ...
    trackbench.tracking.runTracker(dataLog, tracker, false, true, false);

% 6. Check results
disp(trackMetrics)       % posRMS, velRMS, quality
disp(swapReport)         % swapFree, totalSwaps

%%  TRACKER + FILTER COMBOS

Replace it with:

            % Rain degradation (physics-based, range and frequency dependent)
            if enableDegradation && (si.isRadar || si.isIR) && ~si.isMSSR
                sPos_deg = ins.Position(:)';
                try sPos_deg = sPos_deg + si.sensor.MountingLocation(:)'; catch; end

                sDegParams = struct();
                try sDegParams.rangeLimits = si.sensor.RangeLimits; catch; sDegParams.rangeLimits = [0 111120]; end
                try sDegParams.fov = si.sensor.FieldOfView(:)'; catch; sDegParams.fov = [1.4 30]; end

                [pdMult, noiseMult, ~] = trackbench.environment.applyRainDegradation( ...
                    simTime, rainConfig, si, sPos_deg, sDegParams);

                % Drop detections using range-dependent Pd
                if ~isempty(dets)
                    keepDeg = true(numel(dets), 1);
                    for dd = 1:numel(dets)
                        detPos = dets{dd}.Measurement(1:3);
                        slantRange = norm(detPos(:) - sPos_deg(:));
                        pdAtRange = pdMult(slantRange);
                        if rand() > pdAtRange
                            keepDeg(dd) = false;
                        end
                    end
                    dets = dets(keepDeg);
                end

                % Scale measurement noise
                for dd = 1:numel(dets)
